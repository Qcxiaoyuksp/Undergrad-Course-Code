实验2-1 全排列问题
问题描述：设计一个递归算法生成n个元素{r1,r2,…,rn}的全排列。设R={r1,r2,…,rn}是要进行排列的n个元素，Ri=R-{ri}。集合X中元素的全排列记为perm(X)。(ri)perm(X)表示在全排列perm(X)的每一个排列前加上前缀ri得到的排列。
程序框架：
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include "stdlib.h"
#include <iostream>

using namespace std;

void perm(int list[], int k, int m);
void swap(int& a, int& b);

void main()
{
	int a[5] = {0,1,2,3,4};
	perm(a, 0, 3);
	printf("\n");

	system("PAUSE");
}
void perm(int list[], int k, int m)
{
	if (k == m)
	{
		//只剩下一个元素
		for (int i = 0; i <= m; i++)
			cout << list[i];
		cout << endl;
	}
	else
	for (int i = k; i <= m; i++)
	{
		swap(   ,    ); //将第i个元素与第k个元素交换位置
		perm(   ,    ,    ); //递归排列第k+1到第m个元素
		swap(   ,    ); //第i和第k个交换元素的重新换回正常位置
	}
}
void swap(int& a, int& b)
{
	int temp = a;
	a = b;
	b = temp;
}


实验2-2 大整数乘法问题
问题描述：分治法求两个N位的整数的乘积。
程序框架：
#include <math.h>
#include <tchar.h>
#include<iostream>
using namespace std;

int IntegerMultiply(int x, int y, int N)
{
		
	if ((0 == x) || (0 == y))
		return 0;
	if (1 == N)
		return x*y;
	else
	{
		int a = x / (int)pow(10., (int)N / 2);
		int b = x - a * (int)pow(10., N / 2);
		int c = y / (int)pow(10., (int)N / 2);
		int d = y - c * (int)pow(10., N / 2);

		int ac = IntegerMultiply(   ,   ,   );
		int bd = IntegerMultiply(   ,   ,   );
		int ambdmc = IntegerMultiply(   ,   ,   ) + ac + bd;
		return (ac * (int)pow(10., N) + ambdmc * (int)pow(10., N / 2) + bd);
	}
}
int _tmain(int argc, _TCHAR* argv[])
{
	int x = 2304;
	int y = 1520;
	
	cout << "x * y = " << IntegerMultiply(x, y, 4) << endl;
	cout << "x * y = " << x*y << endl;

}
